## 二、数据增强改进

数据增强是小样本场景下**收益最大**的改进方向。遥感图像与自然图像的关键差异在于：

1. **旋转不变性**：目标可以出现在任意角度
2. **尺度差异大**：同一类目标在不同分辨率影像中尺度差异悬殊
3. **大气干扰**：云雾、气溶胶、光照角度
4. **传感器差异**：不同卫星的辐射特性不同

### 2.1 基础几何增强（优先级：🔴 最高）

直接修改 `train.py` 中的 `AUG_CONFIG`，替换为自定义增强配置：

```python
# === 遥感小样本专用增强配置 ===

    "Perspective": {
```

**配置说明**：

| 参数                     | 遥感场景考量                  | 设置值   |
| ------------------------ | ----------------------------- | -------- |
| `Rotate.limit=180`       | 目标可任意朝向，全角度覆盖    | 180°     |
| `Affine.scale=(0.7,1.3)` | 轨道高度变化带来 30% 尺度差异 | 0.7~1.3  |
| `Perspective.scale=0.06` | 大倾角拍摄视角差异            | ≤6% 透视 |
| `GaussianBlur.p=0.15`    | 避免过度模糊影响小目标        | 15%      |
| `GaussNoise.p=0.15`      | 传感器噪声不会太极端          | 15%      |

### 2.2 实例级增强 —— Mosaic / Copy-Paste（优先级：🟡 高）

对于极度小样本场景，可实施实例级合成。由于 RF-DETR 内置不直接支持这类增强，需要在数据预处理或 Dataset 层面实现。

#### 方案 A：数据预处理阶段实现 Mosaic

```python
# tools/mosaic_augmentation.py
"""Mosaic 数据增强预处理脚本 —— 4 张图拼接为 1 张"""

import cv2
import numpy as np
from pathlib import Path
import json
import random


def mosaic_augment(
    images: list[np.ndarray],
    bboxes_list: list[list[list[float]]],
    labels_list: list[list[int]],
    output_size: int = 640,
) -> tuple[np.ndarray, list[list[float]], list[int]]:
    """将 4 张图片拼接为 1 张 Mosaic 增强图像。

    Args:
        images: 4 张图片的列表
        bboxes_list: 每张图片的边界框列表 (xyxy 格式)
        labels_list: 每张图片的标签列表
        output_size: 输出图像尺寸

    Returns:
        (mosaic_image, merged_bboxes, merged_labels)
    """
    h, w = output_size, output_size
    mosaic_image = np.zeros((h, w, 3), dtype=np.uint8)

    # Mosaic 中心点（随机偏移）
    cx = int(random.uniform(w * 0.3, w * 0.7))
    cy = int(random.uniform(h * 0.3, h * 0.7))

    all_bboxes = []
    all_labels = []

    for i, (img, bboxes, labels) in enumerate(zip(images, bboxes_list, labels_list)):
        img_h, img_w = img.shape[:2]

        # 确定放置位置
        if i == 0:  # 左上
            x1a, y1a, x2a, y2a = max(cx - img_w, 0), max(cy - img_h, 0), cx, cy
            x1b, y1b, x2b, y2b = max(img_w - cx, 0), max(img_h - cy, 0), img_w, img_h
        elif i == 1:  # 右上
            x1a, y1a, x2a, y2a = cx, max(cy - img_h, 0), min(cx + img_w, w), cy
            x1b, y1b, x2b, y2b = 0, max(img_h - cy, 0), min(w - cx, img_w), img_h
        elif i == 2:  # 左下
            x1a, y1a, x2a, y2a = max(cx - img_w, 0), cy, cx, min(cy + img_h, h)
            x1b, y1b, x2b, y2b = max(img_w - cx, 0), 0, img_w, min(h - cy, img_h)
        else:  # 右下
            x1a, y1a, x2a, y2a = cx, cy, min(cx + img_w, w), min(cy + img_h, h)
            x1b, y1b, x2b, y2b = 0, 0, min(w - cx, img_w), min(h - cy, img_h)

        # 放置图像
        mosaic_image[y1a:y2a, x1a:x2a] = img[y1b:y2b, x1b:x2b]

        # 调整边界框坐标
        if bboxes:
            bboxes = np.array(bboxes)
            pad_w = x1a - x1b
            pad_h = y1a - y1b
            bboxes[:, [0, 2]] += pad_w
            bboxes[:, [1, 3]] += pad_h
            # 裁剪到图像范围内
            bboxes[:, [0, 2]] = np.clip(bboxes[:, [0, 2]], 0, w)
            bboxes[:, [1, 3]] = np.clip(bboxes[:, [1, 3]], 0, h)
            # 过滤无效框
            valid = (bboxes[:, 2] > bboxes[:, 0]) & (bboxes[:, 3] > bboxes[:, 1])
            all_bboxes.extend(bboxes[valid].tolist())
            all_labels.extend([l for j, l in enumerate(labels) if valid[j]])

    return mosaic_image, all_bboxes, all_labels
```

#### 方案 B：基于数据集统计的 Copy-Paste

对于类别不均衡问题，针对少数类进行实例复制粘贴：

```python
# tools/copy_paste_augmentation.py
"""Copy-Paste 数据增强 —— 将稀有类目标复制到其他图像中"""

import numpy as np
import cv2
import random
from collections import Counter


class CopyPasteAugmenter:
    """基于少数类的 Copy-Paste 增强器。

    统计训练集中各类别样本数，对少数类目标进行跨图像复制增强。
    """

    def __init__(
        self,
        annotations: list[dict],
        class_counts: dict[int, int],
        imbalance_threshold: float = 0.3,
        num_copies: int = 3,
    ):
        """初始化增强器。

        Args:
            annotations: 标注列表 (image_id, bboxes, labels)
            class_counts: 各类别样本数
            imbalance_threshold: 不均衡阈值（低于均值 × 此阈值的类视为少数类）
            num_copies: 每个少数类目标的复制次数
        """
        avg_count = np.mean(list(class_counts.values()))
        self.rare_classes = {cls for cls, count in class_counts.items() if count < avg_count * imbalance_threshold}
        self.num_copies = num_copies
        self.annotations = annotations
        print(f"少数类（需增强）: {self.rare_classes}")

    def paste_instance(
        self,
        src_image: np.ndarray,
        dst_image: np.ndarray,
        bbox: list[float],
        label: int,
    ) -> tuple[np.ndarray, list[float]] | None:
        """将一个目标实例从源图复制到目标图。

        Args:
            src_image: 源图像
            dst_image: 目标图像
            bbox: 目标边界框 (x1, y1, x2, y2)
            label: 类别标签

        Returns:
            (augmented_image, new_bbox) 或 None
        """
        x1, y1, x2, y2 = map(int, bbox)
        instance = src_image[y1:y2, x1:x2].copy()

        if instance.size == 0:
            return None

        h, w = instance.shape[:2]
        dst_h, dst_w = dst_image.shape[:2]

        # 随机放置在目标图像中
        for _ in range(10):  # 最多尝试 10 次
            new_x = random.randint(0, dst_w - w - 1)
            new_y = random.randint(0, dst_h - h - 1)

            # 检查放置区域是否与已有目标重叠过多（简单 IoU 检查）
            if self._check_placement(dst_image[new_y : new_y + h, new_x : new_x + w]):
                dst_image[new_y : new_y + h, new_x : new_x + w] = instance
                return dst_image, [new_x, new_y, new_x + w, new_y + h]

        return None

    def _check_placement(self, region: np.ndarray) -> bool:
        """检查放置区域是否合适（非全黑/全白）。"""
        return region.mean() > 5  # 简单启发式检查
```

## 三、模型架构改进

### 3.1 LoRA 微调编码器（优先级：🔴 最高）

RF-DETR 已内置完整的 LoRA 支持，通过 PEFT 库实现 DoRA（Direction-optimized LoRA）。

**原理**：在 ViT 的 Q/K/V 注意力投影层注入低秩适配器（rank=16），冻结原始权重仅训练适配器参数，大幅减少可训练参数量的同时保持预训练知识。

**代码位置**：[src/rfdetr/models/weights.py:578-623](src/rfdetr/models/weights.py#L578)

**使用方法**：

```python
# 在 train.py 模型构建时添加 backbone_lora=True
model = model_cls(
    num_classes=NUM_CLASSES,
    resolution=default_resolution,
    backbone_lora=True,  # 启用 LoRA（DoRA, rank=16, alpha=16）
    freeze_encoder=False,  # encoder 可训练（仅 LoRA 参数更新）
    pretrain_weights=None,  # COCO 预训练权重通过默认路径加载
)
```

**对小样本的影响**：

- 可训练参数量从 ~40M 降至 ~3M（约减少 90%）
- 过拟合风险大幅降低
- 遥感域适配仅需少量样本

**当 `backbone_lora=True` 且 `freeze_encoder=False` 时的行为**：

- LoRA 适配器注入到 `q_proj`, `v_proj`, `k_proj`, `qkv` 层
- 原始 DINOv2 权重冻结
- 仅 LoRA 参数 + Decoder + Head 参与训练

### 3.2 冻结编码器策略

对于不同数据量级，推荐不同的冻结策略：

| 场景                    | `freeze_encoder` | `backbone_lora` | 说明                |
| ----------------------- | :--------------: | :-------------: | ------------------- |
| 极少量数据（\<200张）   |      `True`      |     `True`      | LoRA + 冻结，最保守 |
| 少量数据（200-1000张）  |     `False`      |     `True`      | LoRA 不冻结，推荐   |
| 中量数据（1000-5000张） |     `False`      |     `True`      | LoRA 不冻结         |
| 充足数据（>5000张）     |     `False`      |     `False`     | 全参数微调          |

### 3.3 多尺度特征金字塔扩展（优先级：🟢 中）

当前 Small 模型仅使用 `projector_scale=["P4"]`（单尺度特征图），这对小目标的检测不利。遥感图像中小目标（如车辆、小型建筑）占比高，需要更丰富的多尺度特征。

#### 方案：扩展至 P3+P4+P5

```python
# 自定义配置类 —— 继承 RFDETRSmallConfig 并扩展特征金字塔
from rfdetr.config import RFDETRSmallConfig, ModelConfig


class RFDETRSmallRemoteConfig(RFDETRSmallConfig):
    """针对遥感小目标的 RF-DETR Small 配置 —— 扩展多尺度金字塔。"""

    # 扩展到三层特征金字塔（原版仅 P4）
    projector_scale: list = ["P3", "P4", "P5"]

    # 使用更浅层的编码器特征（P3 需要更浅层的特征图）
    # Small 默认 out_feature_indexes=[3, 6, 9, 12]
    # 保持不变，MultiScaleProjector 会自动进行上/下采样融合
    out_feature_indexes: list[int] = [3, 6, 9, 12]

    # 保持其他参数不变
    num_windows: int = 2
    dec_layers: int = 3
    patch_size: int = 16
    resolution: int = 512
    positional_encoding_size: int = 32
```

**注意**：

- 扩展 `projector_scale` 会增加计算量约 30-50%
- 需要确保输入分辨率能被 `patch_size * num_windows = 32` 整除
- 使用此配置时需 `pretrain_weights=None` 或加载兼容的 checkpoint
- 如果加载 COCO 预训练权重，P3/P5 投影层将随机初始化

#### 各尺度说明

| 尺度 | 相对 P4 的步长 | 分辨率    | 适用目标大小           |
| ---- | -------------- | --------- | ---------------------- |
| P3   | 2× 放大        | 原图 1/8  | 极小目标（\<16×16 px） |
| P4   | 1×（基准）     | 原图 1/16 | 中小目标（16-64 px）   |
| P5   | 0.5× 缩小      | 原图 1/32 | 大目标（>64 px）       |

**代码位置**：

- 投影器：[src/rfdetr/models/backbone/projector.py](src/rfdetr/models/backbone/projector.py#L188)
- 上采样（P3）：`ConvTranspose2d(stride=2)` — [projector.py:251](src/rfdetr/models/backbone/projector.py#L251)
- 下采样（P5）：`ConvX(stride=2)` — [projector.py:258](src/rfdetr/models/backbone/projector.py#L258)

### 4.2 超参数设计原理

| 超参数          | 默认值 | 建议值 | 设计原理                                       |
| --------------- | ------ | ------ | ---------------------------------------------- |
| `LR`            | 1e-4   | 5e-5   | 小样本下大幅降低学习率，避免过快收敛到局部最优 |
| `LR_ENCODER`    | 1.5e-4 | 5e-5   | 不再对 backbone 使用更高学习率，保持预训练知识 |
| `WEIGHT_DECAY`  | 1e-4   | 5e-4   | 更强的 L2 正则化对抗过拟合                     |
| `CLIP_MAX_NORM` | 0.1    | 0.01   | 更紧的梯度裁剪，防止小 batch 下的梯度噪声      |
| `SMOOTH_ALPHA`  | 0.0    | 0.1    | 标签平滑减少过拟合，让模型不那么"自信"         |
| `WARMUP_EPOCHS` | 0      | 5      | 逐步增加学习率，稳定训练初期                   |
| `LR_DROP`       | 100    | 150    | 推迟降低学习率，给模型更长的学习时间           |
| `EMA_DECAY`     | 0.993  | 0.999  | 更强的指数移动平均，提高模型稳定性             |
| `EPOCHS`        | 100    | 200    | 小样本需要更多迭代才能充分学习                 |

### 5.6 多源数据采样策略

当使用多个数据源训练时，设计合理的采样策略：

```python
# tools/multi_source_sampler.py
"""多数据源加权采样器 —— 在轨数据高权重，公开数据低权重"""

from torch.utils.data import WeightedRandomSampler, ConcatDataset


def build_multi_source_loader(
    in_orbit_dataset,
    public_dataset,
    in_orbit_weight: float = 0.6,
    batch_size: int = 8,
) -> torch.utils.data.DataLoader:
    """构建多数据源加权 DataLoader。

    Args:
        in_orbit_dataset: 在轨测试数据集
        public_dataset: 公开遥感数据集
        in_orbit_weight: 在轨数据的采样权重
        batch_size: 批次大小

    Returns:
        加权采样的 DataLoader
    """
    combined = ConcatDataset([in_orbit_dataset, public_dataset])

    # 为在轨数据分配更高采样权重
    n_in_orbit = len(in_orbit_dataset)
    n_public = len(public_dataset)
    total = n_in_orbit + n_public

    # 计算权重：权重大小反比于数据集大小 × 权重系数
    w_in_orbit = in_orbit_weight / n_in_orbit
    w_public = (1 - in_orbit_weight) / n_public

    weights = [w_in_orbit] * n_in_orbit + [w_public] * n_public

    sampler = WeightedRandomSampler(
        weights=weights,
        num_samples=total,
        replacement=True,
    )

    return torch.utils.data.DataLoader(
        combined,
        batch_size=batch_size,
        sampler=sampler,
        num_workers=8,
    )
```

### 6.2 三阶段训练脚本

```python
# tools/three_stage_training.py
"""三阶段训练流程：COCO 预训练 → 遥感公开数据微调 → 在轨数据精调"""

from rfdetr.variants import RFDETRSmall
from rfdetr import RFDETR
from rfdetr.datasets.aug_configs import (
    AUG_REMOTE_SENSING_AGGRESSIVE,
    AUG_REMOTE_SENSING_FEWSHOT,
)


def stage1_coco_pretrain(num_classes: int) -> RFDETRSmall:
    """阶段 1: 加载 COCO 预训练模型（RF-DETR 官方权重自动下载）。"""
    print("=" * 50)
    print("阶段 1: 加载 COCO 预训练权重")
    print("=" * 50)

    model = RFDETRSmall(
        num_classes=num_classes,
        backbone_lora=True,
        resolution=640,
    )
    # 预训练权重自动下载到 ~/.roboflow/models/
    print("COCO 预训练权重已就绪")
    return model


def stage2_public_remote_sensing(
    model: RFDETRSmall,
    dataset_dir: str,
    num_classes: int,
    output_dir: str = "output/stage2_public",
) -> str:
    """阶段 2: 在公开遥感数据集上微调。"""
    print("\n" + "=" * 50)
    print("阶段 2: 公开遥感数据微调")
    print("=" * 50)

    model.train(
        dataset_dir=dataset_dir,
        dataset_file="roboflow",
        output_dir=output_dir,
        epochs=100,
        batch_size=8,
        lr=1e-4,
        lr_encoder=1e-4,
        lr_scheduler="cosine",
        warmup_epochs=3.0,
        weight_decay=1e-4,
        smooth_alpha=0.05,
        use_ema=True,
        ema_decay=0.995,
        multi_scale=True,
        expanded_scales=True,
        aug_config=AUG_REMOTE_SENSING_AGGRESSIVE,
        early_stopping=True,
        early_stopping_patience=20,
        skip_best_epochs=5,
        device="cuda",
    )

    return f"{output_dir}/checkpoint_best_total.pth"


def stage3_orbital_finetune(
    checkpoint_path: str,
    dataset_dir: str,
    num_classes: int,
    output_dir: str = "output/stage3_orbital",
) -> None:
    """阶段 3: 在轨测试数据精调（核心阶段）。"""
    print("\n" + "=" * 50)
    print("阶段 3: 在轨数据精调")
    print("=" * 50)

    # 从阶段 2 的 checkpoint 加载
    model = RFDETR.from_checkpoint(
        checkpoint_path,
        num_classes=num_classes,
    )

    model.train(
        dataset_dir=dataset_dir,
        dataset_file="roboflow",
        output_dir=output_dir,
        epochs=200,
        batch_size=4,  # 在轨数据少，batch 更小
        grad_accum_steps=8,
        lr=5e-5,  # 极低学习率
        lr_encoder=5e-5,
        lr_scheduler="cosine",
        lr_scheduler_kwargs={"min_factor": 0.01},
        warmup_epochs=5.0,
        weight_decay=5e-4,
        clip_max_norm=0.01,
        smooth_alpha=0.1,
        use_ema=True,
        ema_decay=0.999,
        multi_scale=True,
        expanded_scales=True,
        aug_config=AUG_REMOTE_SENSING_FEWSHOT,
        early_stopping=True,
        early_stopping_patience=30,
        early_stopping_min_delta=0.001,
        early_stopping_use_ema=True,
        skip_best_epochs=10,
        log_per_class_metrics=True,
        device="cuda",
    )

    print(f"\n阶段 3 完成！最终模型保存在: {output_dir}")


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="三阶段遥感目标检测训练")
    parser.add_argument("--num_classes", type=int, default=20, help="类别数")
    parser.add_argument("--public_dataset", type=str, required=False, help="公开遥感数据集路径（阶段 2）")
    parser.add_argument("--orbital_dataset", type=str, required=True, help="在轨测试数据集路径（阶段 3）")
    parser.add_argument("--skip_stage2", action="store_true", help="跳过阶段 2（直接从 COCO 到在轨数据）")
    args = parser.parse_args()

    # 阶段 1: COCO 预训练权重
    model = stage1_coco_pretrain(args.num_classes)

    if args.skip_stage2 or not args.public_dataset:
        print("\n跳过阶段 2，直接从阶段 1 → 阶段 3")
        checkpoint = None
    else:
        checkpoint = stage2_public_remote_sensing(
            model,
            args.public_dataset,
            args.num_classes,
        )

    # 阶段 3: 在轨数据精调
    if checkpoint:
        stage3_orbital_finetune(
            checkpoint,
            args.orbital_dataset,
            args.num_classes,
        )
    else:
        # 直接从阶段 1 模型在轨数据精调
        model.train(
            dataset_dir=args.orbital_dataset,
            dataset_file="roboflow",
            output_dir="output/stage3_orbital_direct",
            epochs=200,
            batch_size=4,
            grad_accum_steps=8,
            lr=5e-5,
            lr_encoder=5e-5,
            lr_scheduler="cosine",
            lr_scheduler_kwargs={"min_factor": 0.01},
            warmup_epochs=5.0,
            weight_decay=5e-4,
            clip_max_norm=0.01,
            smooth_alpha=0.1,
            use_ema=True,
            ema_decay=0.999,
            multi_scale=True,
            expanded_scales=True,
            aug_config=AUG_REMOTE_SENSING_FEWSHOT,
            early_stopping=True,
            early_stopping_patience=30,
            skip_best_epochs=10,
            device="cuda",
        )
```
